// ===== 配置 =====
const cfg = {
  nodeW: 155,
  nodeH: 32,
  hGap: 10,
  vGap: 26,
  padding: 25,
  fontSize: 12,
  rootText: "实验室设备预约与共享管理系统交付项目",
  rootW: 350,
  rootH: 40,
  lv1Row1Count: 4, // 第1层第一行放几个
};

// ===== 颜色 =====
const colors = {
  root: { fill: '#1F4E79', stroke: '#1F4E79', text: '#fff' },
  lv1:  { fill: '#D6E4F0', stroke: '#5B9BD5', text: '#1F4E79' },
  lv2:  { fill: '#F2F7FB', stroke: '#9DC3E6', text: '#333' },
  lv3:  { fill: '#FFFFFF', stroke: '#BFBFBF', text: '#333' },
};

// ===== 构建树 =====
function buildTree() {
  const map = {};
  wbsData.forEach(d => { map[d.code] = { ...d, children: [] }; });
  const roots = [];
  wbsData.forEach(d => {
    if (!d.parent) roots.push(map[d.code]);
    else if (map[d.parent]) map[d.parent].children.push(map[d.code]);
  });
  roots.sort((a, b) => a.code.localeCompare(b.code));
  return roots;
}

// ===== 计算子树宽度 =====
function subW(node) {
  if (!node.children.length) { node._w = cfg.nodeW; return node._w; }
  let t = 0;
  node.children.forEach((c, i) => { t += subW(c); if (i < node.children.length - 1) t += cfg.hGap; });
  node._w = Math.max(cfg.nodeW, t);
  return node._w;
}

// ===== 布局子树 =====
function layout(node, x, y) {
  node._x = x + (node._w - cfg.nodeW) / 2;
  node._y = y;
  if (!node.children.length) return;
  const cy = y + cfg.nodeH + cfg.vGap;
  let cx = x;
  node.children.forEach(c => { layout(c, cx, cy); cx += c._w + cfg.hGap; });
}

// ===== 子树高度 =====
function subH(node) {
  if (!node.children.length) return cfg.nodeH;
  let max = 0;
  node.children.forEach(c => { max = Math.max(max, subH(c)); });
  return cfg.nodeH + cfg.vGap + max;
}

// ===== 生成SVG =====
function genSVG() {
  const roots = buildTree();
  roots.forEach(subW);

  // 分两行
  const row1 = roots.slice(0, cfg.lv1Row1Count);
  const row2 = roots.slice(cfg.lv1Row1Count);

  const rowW = row => {
    let w = 0;
    row.forEach((n, i) => { w += n._w; if (i < row.length - 1) w += cfg.hGap * 2; });
    return w;
  };
  const r1w = rowW(row1), r2w = rowW(row2);
  const maxW = Math.max(r1w, r2w, cfg.rootW);
  const totalW = maxW + cfg.padding * 2;

  const rootY = cfg.padding;
  const r1y = rootY + cfg.rootH + cfg.vGap + 10;

  let r1maxH = 0;
  row1.forEach(n => { r1maxH = Math.max(r1maxH, subH(n)); });

  const r2y = r1y + r1maxH + cfg.vGap + 20;

  let r2maxH = 0;
  row2.forEach(n => { r2maxH = Math.max(r2maxH, subH(n)); });

  const totalH = r2y + r2maxH + cfg.padding;

  let s = `<svg xmlns="http://www.w3.org/2000/svg" width="${totalW}" height="${totalH}" viewBox="0 0 ${totalW} ${totalH}">`;
  s += `<rect width="100%" height="100%" fill="#fff"/>`;

  const rootX = (totalW - cfg.rootW) / 2;
  const r1off = (totalW - r1w) / 2;
  const r2off = (totalW - r2w) / 2;

  // 布局
  let x = r1off;
  row1.forEach(n => { layout(n, x, r1y); x += n._w + cfg.hGap * 2; });
  x = r2off;
  row2.forEach(n => { layout(n, x, r2y); x += n._w + cfg.hGap * 2; });

  // 画子树内部连线
  function lines(node) {
    if (!node.children.length) return;
    const pcx = node._x + cfg.nodeW / 2;
    const pb = node._y + cfg.nodeH;
    node.children.forEach(c => {
      const ccx = c._x + cfg.nodeW / 2;
      const ct = c._y;
      const my = pb + (ct - pb) / 2;
      s += `<path d="M${pcx},${pb} L${pcx},${my} L${ccx},${my} L${ccx},${ct}" fill="none" stroke="#888" stroke-width="1.2"/>`;
      lines(c);
    });
  }
  roots.forEach(lines);

  // 根到第1层连线
  const rcx = rootX + cfg.rootW / 2;
  const rb = rootY + cfg.rootH;

  // 根到第一行
  const r1my = rb + (r1y - rb) / 2;
  row1.forEach(n => {
    const cx = n._x + cfg.nodeW / 2;
    s += `<path d="M${rcx},${rb} L${rcx},${r1my} L${cx},${r1my} L${cx},${r1y}" fill="none" stroke="#888" stroke-width="1.2"/>`;
  });

  // 根到第二行
  const r2my = r1y - cfg.vGap / 2 + (r2y - (r1y - cfg.vGap / 2)) / 2;
  row2.forEach(n => {
    const cx = n._x + cfg.nodeW / 2;
    s += `<path d="M${rcx},${rb} L${rcx},${r2my} L${cx},${r2my} L${cx},${r2y}" fill="none" stroke="#888" stroke-width="1.2"/>`;
  });

  // 画节点
  function draw(node, color, bold) {
    s += `<rect x="${node._x}" y="${node._y}" width="${cfg.nodeW}" height="${cfg.nodeH}" rx="4" ry="4" fill="${color.fill}" stroke="${color.stroke}" stroke-width="1.2"/>`;
    const ty = node._y + cfg.nodeH / 2 + cfg.fontSize / 3;
    const tx = node._x + cfg.nodeW / 2;
    let txt = node.name;
    if (txt.length > 12) txt = txt.substring(0, 12) + '…';
    s += `<text x="${tx}" y="${ty}" text-anchor="middle" fill="${color.text}" font-size="${cfg.fontSize}" font-family="Microsoft YaHei, Arial, sans-serif" font-weight="${bold ? 'bold' : 'normal'}">${txt}</text>`;
    if (node.children.length) {
      const cc = node.level === 1 ? colors.lv2 : colors.lv3;
      node.children.forEach(c => draw(c, cc, false));
    }
  }

  // 根节点
  s += `<rect x="${rootX}" y="${rootY}" width="${cfg.rootW}" height="${cfg.rootH}" rx="4" ry="4" fill="${colors.root.fill}" stroke="${colors.root.stroke}" stroke-width="1.2"/>`;
  s += `<text x="${rootX + cfg.rootW / 2}" y="${rootY + cfg.rootH / 2 + 5}" text-anchor="middle" fill="${colors.root.text}" font-size="14" font-family="Microsoft YaHei, Arial, sans-serif" font-weight="bold">${cfg.rootText}</text>`;

  roots.forEach(n => draw(n, colors.lv1, true));
  s += `</svg>`;
  return s;
}

// 输出
fs.writeFileSync('WBS结构图.svg', genSVG(), 'utf-8');
console.log('已生成：WBS结构图.svg');
